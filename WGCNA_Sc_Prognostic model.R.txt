read_sample <- function(prefix, sample) {
  f <- file.path(paths["single_cell_input"], paste0(prefix,
         c("_matrix.mtx.gz", "_genes.tsv.gz", "_barcodes.tsv.gz")))
  if (!all(file.exists(f))) stop("Missing GSE221575 files for ", sample,
                                ". Extract the GEO RAW archive to data/GSE221575")
  m <- ReadMtx(mtx = f[1], features = f[2], cells = f[3], feature.column = 2)
  if (anyDuplicated(rownames(m))) rownames(m) <- make.unique(rownames(m))
  obj <- CreateSeuratObject(m, project = sample, min.cells = 3, min.features = 200)
  obj$Sample <- sample
  obj$percent.mt <- PercentageFeatureSet(obj, pattern = "^MT-")
  list(obj = obj, raw = ncol(m))
}
loaded <- lapply(seq_len(nrow(sample_info)), function(i)
  read_sample(sample_info$prefix[i], sample_info$sample[i]))
qc_sample <- function(x) {
  obj <- x$obj; initial <- ncol(obj)
  obj <- subset(obj, cells = colnames(obj)[which(obj$nFeature_RNA > 200 &
                        obj$nFeature_RNA < 6000 & obj$percent.mt < 15)])
  after_qc <- ncol(obj)
  if (after_qc < 50) stop("Too few cells after QC")
  sce <- SingleCellExperiment::SingleCellExperiment(
    assays = list(counts = SeuratObject::GetAssayData(obj, assay = "RNA", layer = "counts")))
  sce <- scDblFinder::scDblFinder(sce)
  obj$doublet_score <- sce$scDblFinder.score
  obj$doublet_class <- as.character(sce$scDblFinder.class)
  ndoublet <- sum(obj$doublet_class == "doublet")
  obj <- subset(obj, cells = colnames(obj)[obj$doublet_class == "singlet"])
  list(obj = obj, initial = initial, after_qc = after_qc,
       doublets = ndoublet, singlets = ncol(obj), raw = x$raw)
}

  dplyr::mutate(GDCquery_clinic(project = "TCGA-COAD", type = "clinical"), project = "TCGA-COAD"),

  if ("tpm_unstranded" %in% assayNames(se)) {
    tpm_assay <- "tpm_unstranded"
  } else stop("TPM assay not found; inspect assayNames(se)")

               dimnames = list(patient_id, hub_genes))

  idx <- which(feature_symbols == g)
  if (length(idx) != 1) stop("Expected exactly one row for ", g, "; found ", length(idx))
  expr[, g] <- as.numeric(tpm[idx, ])
}
expr <- log2(expr + 1) # verify against original saved tcga_model_data preprocessing
clinical <- clinical[!duplicated(clinical$submitter_id), ]
match_idx <- match(rownames(expr), clinical$submitter_id)
model_data <- cbind(data.frame(patient_id = rownames(expr), expr, check.names = FALSE),
                    clinical[match_idx, setdiff(names(clinical), "submitter_id"), drop = FALSE])
model_data$OS_event <- as.integer(tolower(as.character(model_data$vital_status)) == "dead")
model_data$OS_time <- ifelse(model_data$OS_event == 1,
                             suppressWarnings(as.numeric(model_data$days_to_death)),

# 7. 10-FOLD LASSO-COX, COEFFICIENTS, RISK GROUPS, ROC
x <- as.matrix(model_data[, hub_genes]); storage.mode(x) <- "double"
x_scaled <- scale(x)
scaling <- list(center = attr(x_scaled, "scaled:center"),
                scale = attr(x_scaled, "scaled:scale"))
set.seed(12345)
cv_lasso <- cv.glmnet(x_scaled, Surv(model_data$OS_time, model_data$OS_event),
                      family = "cox", alpha = 1, nfolds = 10,
                      standardize = FALSE, type.measure = "deviance")
coef_mat <- as.matrix(coef(cv_lasso, s = "lambda.min"))
coefs <- data.frame(Gene = rownames(coef_mat), Coefficient = coef_mat[, 1])
write.csv(coefs, file.path(paths["survival_output"], "LASSO_coefficients.csv"), row.names = FALSE)
write.csv(data.frame(lambda_min = cv_lasso$lambda.min, lambda_1se = cv_lasso$lambda.1se),
          file.path(paths["survival_output"], "LASSO_lambda.csv"), row.names = FALSE)
pdf(file.path(paths["figures"], "LASSO_cross_validation.pdf")); plot(cv_lasso); dev.off()
model_data$risk_score <- as.numeric(predict(cv_lasso, newx = x_scaled,
                                             s = "lambda.min", type = "link"))
model_data$risk_group <- factor(ifelse(model_data$risk_score >= median(model_data$risk_score),
                                        "High", "Low"), levels = c("Low", "High"))
cox_cont <- coxph(Surv(OS_time, OS_event) ~ risk_score, data = model_data)
cox_group <- coxph(Surv(OS_time, OS_event) ~ risk_group, data = model_data)
model_data$risk_score_SD <- as.numeric(scale(model_data$risk_score))
cox_SD <- coxph(Surv(OS_time, OS_event) ~ risk_score_SD, data = model_data)
sink(file.path(paths["survival_output"], "Cox_models.txt"))
print(summary(cox_cont)); print(summary(cox_group)); print(summary(cox_SD)); sink()
pdf(file.path(paths["figures"], "Kaplan_Meier_risk_groups.pdf"), width = 8, height = 6)
plot(survfit(Surv(OS_time, OS_event) ~ risk_group, data = model_data),
     col = c("steelblue", "firebrick"), xlab = "Days", ylab = "Overall survival")
legend("bottomleft", legend = levels(model_data$risk_group),
       col = c("steelblue", "firebrick"), lty = 1)
dev.off()
roc <- timeROC::timeROC(T = model_data$OS_time, delta = model_data$OS_event,
                        marker = model_data$risk_score, cause = 1,
                        weighting = "marginal", times = c(365, 1095, 1825), iid = TRUE)

                      data = validation)
